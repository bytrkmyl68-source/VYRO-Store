const PRODUCT_KEY = 'vyro_products'; const ORDER_KEY = 'vyro_orders'; function money(value) { return `${Number(value || 0).toLocaleString('ar-EG')} ج.م`; } function getProducts() { try { const products = JSON.parse(localStorage.getItem(PRODUCT_KEY) || '[]'); return Array.isArray(products) ? products : []; } catch { return []; } } function getOrders() { try { const orders = JSON.parse(localStorage.getItem(ORDER_KEY) || '[]'); return Array.isArray(orders) ? orders : []; } catch { return []; } } function seedProducts() { if (getProducts().length) return; const starter = [ { id: crypto.randomUUID(), name: 'تيشيرت كلاسيك', category: 'basic', price: 299, stock: 12, active: true, image: '' }, { id: crypto.randomUUID(), name: 'جاكيت اوفر سايز', category: 'oversize', price: 499, stock: 8, active: true, image: '' }, { id: crypto.randomUUID(), name: 'هودي أساسي', category: 'basic', price: 349, stock: 15, active: true, image: '' } ]; localStorage.setItem(PRODUCT_KEY, JSON.stringify(starter)); } function createOrder(product) { const orders = getOrders(); const newOrder = { id: `#${Date.now()}`, customer: 'عميل جديد', date: new Date().toISOString().slice(0, 10), total: Number(product.price || 0), status: 'قيد التنفيذ' }; orders.push(newOrder); localStorage.setItem(ORDER_KEY, JSON.stringify(orders)); } function renderProducts() { const products = getProducts().filter((product) => product.active !== false); const target = document.querySelector('[data-products]'); if (!target) return; if (!products.length) { target.innerHTML = '
لا توجد منتجات متاحة حالياً
'; return; } target.innerHTML = products .map((product) => { const image = product.image || 'https://images.unsplash.com/photo-1521572267360-ee0c2909d518?auto=format&fit=crop&w=800&q=80'; return `
${product.category === 'oversize' ? 'أوفر سايز' : 'أساسي'}

${product.name}

${product.stock} في المخزن ${money(product.price)}
`; }) .join(''); document.querySelectorAll('.buy-btn').forEach((button) => { button.addEventListener('click', () => { const product = products.find((item) => item.id === button.dataset.productId); if (!product) return; createOrder(product); alert('تم إنشاء الطلب بنجاح'); }); }); } document.addEventListener('DOMContentLoaded', () => { seedProducts(); renderProducts(); });
